## 🎯 USAGE EXAMPLES

### Complete Knowledge Graph Workflow
```python
from knowledge_engine.engine import KnowledgeEngine

# Initialize with all features
engine = KnowledgeEngine()

# 1. Add knowledge with temporal tracking
await engine.add_knowledge_temporal(
    content="Python async/await improves I/O performance",
    valid_at=datetime(2024, 1, 1)
)

# 2. Extract with quality enhancement
result = await engine.extract_with_quality(
    text="Django framework simplifies web development...",
    schema="software_engineering",
    enable_enhancement=True
)
print(f"Quality: {result['quality_score']['overall']:.2f}")

# 3. Process with AIKG pipeline
aikg_result = await engine.process_with_aikg(
    text="Machine learning algorithms analyze data...",
    enable_standardization=True,
    enable_inference=True,
    generate_visualization=True
)
print(f"Entities standardized: {aikg_result['entity_reduction_rate']:.1%}")
print(f"Relationships inferred: {aikg_result['inferred_triple_count']}")

# 4. Analyze with KarateClub
structure = await engine.analyze_with_karateclub(
    analysis_type="structure",
    target=knowledge_graph
)
print(f"Communities: {len(structure['communities'])}")

# 5. Search with hybrid algorithm
results = await engine.search_with_graphiti(
    query="web development frameworks",
    use_hybrid=True,
    rerank_method="rrf"
)

# 6. Get similar knowledge
similar = await engine.get_similar_knowledge(
    query="async programming",
    top_k=10
)
```
