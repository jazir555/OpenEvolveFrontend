export interface JupyterNotebook {
  cells: JupyterCell[];
  metadata: Record<string, any>;
  nbformat: number;
  nbformat_minor: number;
}

export interface JupyterCell {
  cell_type: 'code' | 'markdown' | 'raw';
  source: string[];
  metadata: Record<string, any>;
  execution_count?: number;
  outputs?: JupyterOutput[];
}

export interface JupyterOutput {
  output_type: 'stream' | 'execute_result' | 'display_data' | 'error';
  text?: string;
  data?: Record<string, any>;
  ename?: string;
  evalue?: string;
  traceback?: string[];
}

export class JupyterNotebookExecutor {
  async executeNotebook(
    notebook: JupyterNotebook,
    options: ExecutionOptions
  ): Promise<ExecutionResult> {
    const results: CellResult[] = [];

    for (const cell of notebook.cells) {
      if (cell.cell_type === 'code') {
        const result = await this.executeCell(cell, options);
        results.push(result);

        // Stop if cell failed
        if (!result.success) {
          break;
        }
      }
    }

    return {
      success: results.every(r => r.success),
      cellResults: results,
      totalExecutionTime: results.reduce((sum, r) => sum + r.executionTime, 0),
    };
  }

  async executeCell(
    cell: JupyterCell,
    options: ExecutionOptions
  ): Promise<CellResult> {
    const code = cell.source.join('\n');
    const startTime = Date.now();

    try {
      const result = await this.executeInSandbox({
        command: 'python',
        args: ['-c', code],
        timeout: options.timeout || 30000,
      });

      const executionTime = Date.now() - startTime;

      return {
        cellId: this.generateCellId(cell),
        success: result.exitCode === 0,
        output: result.stdout,
        error: result.stderr,
        executionCount: (cell.execution_count || 0) + 1,
        executionTime,
        outputs: this.parseOutputs(result.stdout, result.stderr),
      };
    } catch (error) {
      return {
        cellId: this.generateCellId(cell),
        success: false,
        error: error.message,
        executionTime: Date.now() - startTime,
      };
    }
  }

  async convertNotebookToScript(notebook: JupyterNotebook): Promise<string> {
    const lines: string[] = [];

    for (const cell of notebook.cells) {
      if (cell.cell_type === 'code') {
        lines.push(...cell.source);
        lines.push(''); // Empty line between cells
      } else if (cell.cell_type === 'markdown') {
        lines.push('#', ...cell.source);
        lines.push('');
      }
    }

    return lines.join('\n');
  }

  async convertScriptToNotebook(script: string): Promise<JupyterNotebook> {
    const lines = script.split('\n');
    const cells: JupyterCell[] = [];
    let currentCell: string[] = [];

    for (const line of lines) {
      if (line.startsWith('# %%')) {
        // New cell marker
        if (currentCell.length > 0) {
          cells.push({
            cell_type: 'code',
            source: currentCell,
            metadata: {},
          });
          currentCell = [];
        }
      } else {
        currentCell.push(line);
      }
    }

    // Add last cell
    if (currentCell.length > 0) {
      cells.push({
        cell_type: 'code',
        source: currentCell,
        metadata: {},
      });
    }

    return {
      cells,
      metadata: {
        kernelspec: {
          display_name: 'Python 3',
          language: 'python',
          name: 'python3',
        },
      },
      nbformat: 4,
      nbformat_minor: 4,
    };
  }

import json

}

`;

  private generateFilters(filters: Filter[]): string {
    const conditions: string[] = [];

    for (const filter of filters) {
      switch (filter.operator) {
        case 'eq':
          conditions.push(`${filter.field} == ${JSON.stringify(filter.value)}`);
          break;
        case 'ne':
          conditions.push(`${filter.field} != ${JSON.stringify(filter.value)}`);
          break;
        case 'gt':
          conditions.push(`${filter.field} > ${filter.value}`);
          break;
        case 'gte':
          conditions.push(`${filter.field} >= ${filter.value}`);
          break;
        case 'lt':
          conditions.push(`${filter.field} < ${filter.value}`);
          break;
        case 'lte':
          conditions.push(`${filter.field} <= ${filter.value}`);
          break;
        case 'in':
          conditions.push(`${filter.field}.in_(${JSON.stringify(filter.value)})`);
          break;
        case 'like':
          conditions.push(`${filter.field}.like('${filter.value}')`);
          break;
        case 'ilike':
          conditions.push(`${filter.field}.ilike('${filter.value}')`);
          break;
        case 'is_null':
          conditions.push(`${filter.field} == None`);
          break;
        case 'is_not_null':
          conditions.push(`${filter.field} != None`);
          break;
        case 'between':
          conditions.push(`${filter.field}.between(${filter.value[0]}, ${filter.value[1]})`);
          break;
      }
    }

    if (filters.logic === 'or') {
      return `q = q.filter(or_(${conditions.join(', ')}))`;
    } else {
      return `q = q.filter(and_(${conditions.join(', ')}))`;
    }
  }

export class JupyterNotebookExecutor {
  async executeNotebook(
    notebook: JupyterNotebook,
    options: ExecutionOptions
  ): Promise<ExecutionResult> {
    const results: CellResult[] = [];

    for (const cell of notebook.cells) {
      if (cell.cell_type === 'code') {
        const result = await this.executeCell(cell, options);
        results.push(result);

        // Stop if cell failed
        if (!result.success) {
          break;
        }
      }
    }

    return {
      success: results.every(r => r.success),
      cellResults: results,
      totalExecutionTime: results.reduce((sum, r) => sum + r.executionTime, 0),
    };
  }

  async executeCell(
    cell: JupyterCell,
    options: ExecutionOptions
  ): Promise<CellResult> {
    const code = cell.source.join('\n');
    const startTime = Date.now();

    try {
      const result = await this.executeInSandbox({
        command: 'python',
        args: ['-c', code],
        timeout: options.timeout || 30000,
      });

      const executionTime = Date.now() - startTime;

      return {
        cellId: this.generateCellId(cell),
        success: result.exitCode === 0,
        output: result.stdout,
        error: result.stderr,
        executionCount: (cell.execution_count || 0) + 1,
        executionTime,
        outputs: this.parseOutputs(result.stdout, result.stderr),
      };
    } catch (error) {
      return {
        cellId: this.generateCellId(cell),
        success: false,
        error: error.message,
        executionTime: Date.now() - startTime,
      };
    }
  }

  async convertNotebookToScript(notebook: JupyterNotebook): Promise<string> {
    const lines: string[] = [];

    for (const cell of notebook.cells) {
      if (cell.cell_type === 'code') {
        lines.push(...cell.source);
        lines.push(''); // Empty line between cells
      } else if (cell.cell_type === 'markdown') {
        lines.push('#', ...cell.source);
        lines.push('');
      }
    }

    return lines.join('\n');
  }

  async convertScriptToNotebook(script: string): Promise<JupyterNotebook> {
    const lines = script.split('\n');
    const cells: JupyterCell[] = [];
    let currentCell: string[] = [];

    for (const line of lines) {
      if (line.startsWith('# %%')) {
        // New cell marker
        if (currentCell.length > 0) {
          cells.push({
            cell_type: 'code',
            source: currentCell,
            metadata: {},
          });
          currentCell = [];
        }
      } else {
        currentCell.push(line);
      }
    }

    // Add last cell
    if (currentCell.length > 0) {
      cells.push({
        cell_type: 'code',
        source: currentCell,
        metadata: {},
      });
    }

    return {
      cells,
      metadata: {
        kernelspec: {
          display_name: 'Python 3',
          language: 'python',
          name: 'python3',
        },
      },
      nbformat: 4,
      nbformat_minor: 4,
    };
  }

  private generateFilters(filters: Filter[]): string {
    const conditions: string[] = [];

    for (const filter of filters) {
      switch (filter.operator) {
        case 'eq':
          conditions.push(`${filter.field} == ${JSON.stringify(filter.value)}`);
          break;
        case 'ne':
          conditions.push(`${filter.field} != ${JSON.stringify(filter.value)}`);
          break;
        case 'gt':
          conditions.push(`${filter.field} > ${filter.value}`);
          break;
        case 'gte':
          conditions.push(`${filter.field} >= ${filter.value}`);
          break;
        case 'lt':
          conditions.push(`${filter.field} < ${filter.value}`);
          break;
        case 'lte':
          conditions.push(`${filter.field} <= ${filter.value}`);
          break;
        case 'in':

        case 'is_not_null':
          conditions.push(`${filter.field} != None`);
          break;
        case 'between':
          conditions.push(`${filter.field}.between(${filter.value[0]}, ${filter.value[1]})`);
          break;
      }
    }

    if (filters.logic === 'or') {
      return `q = q.filter(or_(${conditions.join(', ')}))`;
    } else {
      return `q = q.filter(and_(${conditions.join(', ')}))`;
    }
