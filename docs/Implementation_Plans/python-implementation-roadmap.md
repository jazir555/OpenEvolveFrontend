// src/lib/python/async/async-executor.ts

export class AsyncPythonExecutor {
  async executeAsync(code: string): Promise<ExecutionResult> {
    const wrappedCode = `
import asyncio
import sys
from io import StringIO

async def _main():
    # Capture output
    old_stdout = sys.stdout
    sys.stdout = StringIO()

    try:
        # User code
        ${code.split('\n').map(line => '        ' + line).join('\n')}

        # Get output
        output = sys.stdout.getvalue()
        return {'success': True, 'output': output}
    except Exception as e:
        return {'success': False, 'error': str(e)}
    finally:
        sys.stdout = old_stdout

# Run async code
result = asyncio.run(_main())
print(result)
`;

    const result = await this.executeInSandbox({
      command: 'python',
      args: ['-c', wrappedCode],
      timeout: 30000,
    });

    return JSON.parse(result.stdout);
  }

  async executeConcurrent(
    tasks: AsyncTask[]
  ): Promise<ConcurrentExecutionResult> {
    const code = `
import asyncio
import json

async def task_${tasks[0].id}():
    ${tasks[0].code}

${tasks.map(task => `
async def task_${task.id}():
    ${task.code}
`).join('\n')}

async def main():
    results = await asyncio.gather(
${tasks.map(task => `        task_${task.id}(),`).join('\n')}
        return_exceptions=True
    )

    # Process results
    output = []
    for i, result in enumerate(results):
        if isinstance(result, Exception):
            output.append({
                'task_id': '${tasks[i].id}',
                'success': False,
                'error': str(result)
            })
        else:
            output.append({
                'task_id': '${tasks[i].id}',
                'success': True,
                'result': result
            })

    print(json.dumps(output))

asyncio.run(main())
`;

    const result = await this.executeInSandbox({
      command: 'python',
      args: ['-c', code],
      timeout: tasks.reduce((sum, task) => sum + (task.timeout || 10000), 0),
    });

    return {
      results: JSON.parse(result.stdout),
      totalExecutionTime: result.executionTime,
    };
  }
}

export class DockerfileGenerator {
  async generate(project: PythonProject): Promise<Dockerfile> {
    const dockerfile = new DockerfileBuilder();

    // Base image
    dockerfile.from(`python:${project.pythonVersion}-slim`);

    // Set environment variables
    dockerfile.env('PYTHONUNBUFFERED', '1');
    dockerfile.env('PYTHONDONTWRITEBYTECODE', '1');
    dockerfile.env('PYTHONPATH', '/app');

    // Set work directory
    dockerfile.workdir('/app');

    // Install system dependencies
    dockerfile.run(`
      apt-get update && \\
      apt-get install -y --no-install-recommends \\
        build-essential \\
        gcc \\
        && rm -rf /var/lib/apt/lists/*
    `);

    // Install Python dependencies
    if (project.requirementsFile) {
      dockerfile.copy(`${project.requirementsFile}`, '/app/');
      dockerfile.run('pip install --no-cache-dir -r requirements.txt');
    }

    if (project.packageManager === 'poetry') {
      dockerfile.copy('pyproject.toml', 'poetry.lock', '/app/');
      dockerfile.run('pip install poetry && poetry install --no-dev');
    }

    // Copy application code
    dockerfile.copy('.', '/app/');

    // Create non-root user
    dockerfile.run('useradd -m -u 1000 appuser');
    dockerfile.user('appuser');

    // Expose port
    if (project.port) {
      dockerfile.expose(project.port);
    }

    // Health check
    if (project.healthCheck) {
      dockerfile.healthcheck(
        `CMD curl -f http://localhost:${project.port}/health || exit 1`,
        { interval: '30s', timeout: '10s', retries: 3 }
      );
    }

    // Run command
    dockerfile.cmd(project.startCommand || 'python main.py');

    return dockerfile.toString();
  }

  async generateMultiStage(project: PythonProject): Promise<string> {
    return `
# Builder stage
FROM python:${project.pythonVersion}-slim as builder

WORKDIR /app

# Install build dependencies
RUN apt-get update && \\
    apt-get install -y --no-install-recommends \\
        build-essential \\
        gcc \\
    && rm -rf /var/lib/apt/lists/*

# Install Python dependencies
COPY ${project.requirementsFile || 'requirements.txt'} .
RUN pip install --user --no-cache-dir -r ${project.requirementsFile || 'requirements.txt'}

# Runtime stage
FROM python:${project.pythonVersion}-slim

WORKDIR /app

# Copy dependencies from builder
COPY --from=builder /root/.local /root/.local

# Copy application
COPY . .

# Make sure scripts in .local are usable
ENV PATH=/root/.local/bin:$PATH

# Set environment
ENV PYTHONUNBUFFERED=1 \\
    PYTHONDONTWRITEBYTECODE=1 \\
    PYTHONPATH=/app

# Create non-root user
RUN useradd -m -u 1000 appuser
USER appuser

# Expose port
${project.port ? `EXPOSE ${project.port}` : ''}

# Health check
${project.healthCheck ? `HEALTHCHECK CMD curl -f http://localhost:${project.port}/health || exit 1` : ''}

# Run application
CMD ${project.startCommand || 'python main.py'}
`;
  }
}

// src/lib/python/profiling/profiler.ts

export class PythonProfiler {
  async profileExecution(
    code: string,
    profilerType: 'cprofile' | 'line_profiler' | 'memory_profiler'
  ): Promise<ProfileResult> {
    switch (profilerType) {
      case 'cprofile':
        return this.profileWithcProfile(code);
      case 'line_profiler':
        return this.profileWithLineProfiler(code);
      case 'memory_profiler':
        return this.profileMemory(code);
    }
  }

  private async profileWithcProfile(code: string): Promise<ProfileResult> {
    const wrappedCode = `
import cProfile
import pstats
import io
import json

# Wrap code for profiling
pr = cProfile.Profile()
pr.enable()

# User code
${code}

pr.disable()

# Get stats
s = io.StringIO()
ps = pstats.Stats(pr, stream=s).sort_stats('cumulative')
ps.print_stats()

# Parse stats
stats_text = s.getvalue()
lines = stats_text.split('\\n')

# Extract function calls
function_calls = []
for line in lines[5:]:  # Skip header
    parts = line.split()
    if len(parts) >= 6:
        try:
            function_calls.append({
                'ncalls': parts[0],
                'tottime': float(parts[1]),
                'cumtime': float(parts[3],
                'filename': parts[-1],
                'lineno': int(parts[-2]),
                'function': parts[-3]
            })
        except:
            pass

result = {
    'total_calls': pr.get_total_calls(),
    'primitive_calls': 0,  # Would need more parsing
    'functions': function_calls[:20]  # Top 20
}

print(json.dumps(result))
`;

    const result = await this.executeInSandbox({
      command: 'python',
      args: ['-c', wrappedCode],
      timeout: 60000,
    });

    return JSON.parse(result.stdout);
  }

  private async profileWithLineProfiler(code: string): Promise<ProfileResult> {
    // First, write code to file
    await this.writeTempFile(code, 'profile_me.py');

    const profileCode = `
import line_profiler
import json
import sys

lp = line_profiler.LineProfiler()

# Read and execute the code with line profiler
exec(open('/workspace/profile_me.py').read(), lp.__dict__)

# Print stats
import io
s = io.StringIO()
lp.print_stats(stream=s)

# Parse line-by-line timings
lines = s.getvalue().split('\\n')
line_timings = []

for line in lines:
    if '%' in line and 'us' in line:
        parts = line.split()
        if len(parts) >= 6:
            try:
                line_timings.append({
                    'line_number': int(parts[0]),
                    'hits': int(parts[1]),
                    'time': float(parts[2]),
                    'per_hit': float(parts[3]),
                    'percent': float(parts[4]),
                    'line': ' '.join(parts[6:])
                })
            except:
                pass

result = {
    'line_timings': line_timings
}

print(json.dumps(result))
`;

    const result = await this.executeInSandbox({
      command: 'python',
      args: ['-m', 'line_profiler', profileCode],
      timeout: 60000,
    });

    return JSON.parse(result.stdout);
  }

  private async profileMemory(code: string): Promise<ProfileResult> {
    const wrappedCode = `
from memory_profiler import profile
import json

@profile(precision=4)
def profiled_function():
${code.split('\n').map(line => '    ' + line).join('\n')}

# Run and capture output
import io
import sys
old_stdout = sys.stdout
sys.stdout = io.StringIO()

profiled_function()

output = sys.stdout.getvalue()
sys.stdout = old_stdout

# Parse memory usage
lines = output.split('\\n')
memory_usage = []

for line in lines:
    if 'MiB' in line:
        parts = line.split()
        if len(parts) >= 4:
            try:
                memory_usage.append({
                    'line': parts[0],
                    'memory_usage': float(parts[1]),
                    'increment': float(parts[3]),
                    'occurrences': parts[4]
                })
            except:
                pass

result = {
    'memory_usage': memory_usage
}

print(json.dumps(result))
`;

    const result = await this.executeInSandbox({
      command: 'python',
      args: ['-m', 'memory_profiler', wrappedCode],
      timeout: 60000,
    });

    return JSON.parse(result.stdout);
  }

  async benchmark(
    code: string,
    iterations: number = 1000
  ): Promise<BenchmarkResult> {
    const benchmarkCode = `
import timeit
import json

setup = """
${code}
"""

stmt = """
# Code to benchmark
"""

# Time it
timer = timeit.Timer(stmt, setup)
times = timer.repeat(repeat=${iterations}, number=1)

result = {
    'iterations': ${iterations},
    'total_time': sum(times),
    'average_time': sum(times) / len(times),
    'min_time': min(times),
    'max_time': max(times),
    'std_dev': (sum((t - sum(times)/len(times))**2 for t in times) / len(times))**0.5
}

print(json.dumps(result))
`;

    const result = await this.executeInSandbox({
      command: 'python',
      args: ['-c', benchmarkCode],
      timeout: 120000,
    });

    return JSON.parse(result.stdout);
  }

export class AsyncPythonExecutor {
  async executeAsync(code: string): Promise<ExecutionResult> {
    const wrappedCode = `
import asyncio
import sys
from io import StringIO

async def _main():
    # Capture output
    old_stdout = sys.stdout
    sys.stdout = StringIO()

    try:
        # User code
        ${code.split('\n').map(line => '        ' + line).join('\n')}

        # Get output
        output = sys.stdout.getvalue()
        return {'success': True, 'output': output}
    except Exception as e:
        return {'success': False, 'error': str(e)}
    finally:
        sys.stdout = old_stdout

# Run async code
result = asyncio.run(_main())
print(result)
`;

    const result = await this.executeInSandbox({
      command: 'python',
      args: ['-c', wrappedCode],
      timeout: 30000,
    });

    return JSON.parse(result.stdout);
  }

  async executeConcurrent(
    tasks: AsyncTask[]
  ): Promise<ConcurrentExecutionResult> {
    const code = `
import asyncio
import json

async def task_${tasks[0].id}():
    ${tasks[0].code}

${tasks.map(task => `
async def task_${task.id}():
    ${task.code}
`).join('\n')}

async def main():
    results = await asyncio.gather(
${tasks.map(task => `        task_${task.id}(),`).join('\n')}
        return_exceptions=True
    )

    # Process results
    output = []
    for i, result in enumerate(results):
        if isinstance(result, Exception):
            output.append({
                'task_id': '${tasks[i].id}',
                'success': False,
                'error': str(result)
            })
        else:
            output.append({
                'task_id': '${tasks[i].id}',
                'success': True,
                'result': result
            })

    print(json.dumps(output))

asyncio.run(main())
`;

    const result = await this.executeInSandbox({
      command: 'python',
      args: ['-c', code],
      timeout: tasks.reduce((sum, task) => sum + (task.timeout || 10000), 0),
    });

    return {
      results: JSON.parse(result.stdout),
      totalExecutionTime: result.executionTime,
    };
  }
}

export class DockerfileGenerator {
  async generate(project: PythonProject): Promise<Dockerfile> {
    const dockerfile = new DockerfileBuilder();

    // Base image
    dockerfile.from(`python:${project.pythonVersion}-slim`);

    // Set environment variables
    dockerfile.env('PYTHONUNBUFFERED', '1');
    dockerfile.env('PYTHONDONTWRITEBYTECODE', '1');
    dockerfile.env('PYTHONPATH', '/app');

    // Set work directory
    dockerfile.workdir('/app');

    // Install system dependencies
    dockerfile.run(`
      apt-get update && \\
      apt-get install -y --no-install-recommends \\
        build-essential \\
        gcc \\
        && rm -rf /var/lib/apt/lists/*
    `);

    // Install Python dependencies
    if (project.requirementsFile) {
      dockerfile.copy(`${project.requirementsFile}`, '/app/');
      dockerfile.run('pip install --no-cache-dir -r requirements.txt');
    }

    if (project.packageManager === 'poetry') {
      dockerfile.copy('pyproject.toml', 'poetry.lock', '/app/');
      dockerfile.run('pip install poetry && poetry install --no-dev');
    }

    // Copy application code
    dockerfile.copy('.', '/app/');

    // Create non-root user
    dockerfile.run('useradd -m -u 1000 appuser');
    dockerfile.user('appuser');

    // Expose port
    if (project.port) {
      dockerfile.expose(project.port);
    }

    // Health check
    if (project.healthCheck) {
      dockerfile.healthcheck(
        `CMD curl -f http://localhost:${project.port}/health || exit 1`,
        { interval: '30s', timeout: '10s', retries: 3 }
      );
    }

    // Run command
    dockerfile.cmd(project.startCommand || 'python main.py');

    return dockerfile.toString();
  }

print(json.dumps(result))

# Code to benchmark

    'iterations': ${iterations},
    'total_time': sum(times),
    'average_time': sum(times) / len(times),
    'min_time': min(times),

}

print(json.dumps(result))
`;

    const result = await this.executeInSandbox({
