```typescript
export async function exportConfiguration() {
    const config: ExportedConfig = { /* ... */ };
    
    try {
        // MessagePack + compression
        const blob = await serialize(config, {
            format: 'msgpack',
            compress: true,
            onProgress: (percent) => console.log(`Export: ${percent}%`)
        });
        
        downloadBlob(blob, `config-${timestamp}.msgpack.gz`);
        console.log(`Exported: ${formatBytes(blob.size)} (compressed)`);
    } catch (error) {
        // Fallback to JSON
        const configString = JSON.stringify(config, null, 2);
        // ... JSON export code
    }
}
```

### Import Function (ConfigManager.ts)

```typescript
export async function handleImportConfiguration(event: Event) {
    const file = event.target.files[0];
    
    try {
        // Auto-detect format
        const imported = await deserialize(file, (p) => {
            console.log(`Import: ${p}%`);
        });
        
        // Handle versioned or legacy
        const configToRestore = imported._version ? imported.data : imported;
        
        // Sanitize state
        const sanitized = sanitizeState(configToRestore);
        
        // Restore configuration
        await restoreConfiguration(sanitized);
    } catch (error) {
        alert(`Import failed: ${error.message}`);
    }
}
```
