## 🚀 HOW TO USE THE UNIFIED PLUGIN

### Option 1: Direct Integration (Recommended)

```typescript
// In BubbleLab or any React app
import { OpenEvolvePlugin } from '@openevolve/plugin';
import { registerPlugin } from '@bubblelabs/studio';

// Register the plugin
registerPlugin(OpenEvolvePlugin);

// Access all nodes
import {
  DecompositionNode,
  SolutionNode,
  VerificationNode,
  EvolutionNode,
  AdversarialNode,
  KnowledgeQueryNode,
  LeanAideNode,
  HephaestusNode,
  MDAPNode,
  MAKERNode
} from '@openevolve/plugin/nodes';

// Access all components
import {
  EvolutionConfigPanel,
  AdversarialConfigPanel,
  DecompositionConfigPanel,
  IntegrationConfigPanel,
  OpenEvolveDashboard,
  AnalyticsDashboard
} from '@openevolve/plugin/components';

// Access all hooks
import {
  useEvolution,
  useAdversarial,
  useDecomposition,
  useKnowledgeEngine,
  useLeanAIDE,
  useHephaestus
} from '@openevolve/plugin/hooks';
```

### Option 2: Node Factory (Dynamic Creation)

```typescript
import { getNode, searchNodes, getCategories } from '@openevolve/plugin/nodes';

// Get a specific node
const evolutionNode = getNode('Evolution', 'my-evolution-node', {
  config: {
    populationSize: 100,
    generations: 50,
    mutationRate: 0.1
  }
});

// Search for nodes
const workflowNodes = searchNodes('workflow');

// Get all categories
const categories = getCategories();
// ['workflow', 'verification', 'knowledge', 'generation', 'planning']
```

### Option 3: Enhanced Plugin Factory

```typescript
import { createEnhancedOpenEvolvePlugin } from '@openevolve/plugin';

const plugin = createEnhancedOpenEvolvePlugin({
  enablePerformanceProfiling: true,
  enableSecurityAudit: true,
  integrations: {
    leanAIDE: { enabled: true },
    hephaestus: { enabled: true },
    mdap: { enabled: true }
  }
});

// Execute with enhanced features
const result = await plugin.executeEvolutionWithEnhancedFeatures(
  'Solve complex problem',
  {
    performanceProfile: 'high-performance',
    securityLevel: 'strict'
  }
);
```
